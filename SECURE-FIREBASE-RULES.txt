🔐 SECURE FIREBASE RULES FOR PRODUCTION 🔐

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Users can read and write their own user documents
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // Activities collection - Secure rules
    match /activities/{activityId} {
      // Anyone authenticated can read activities
      allow read: if request.auth != null;
      
      // Only authenticated users can create activities
      allow create: if request.auth != null 
        && request.auth.uid == request.resource.data.organizerId;
      
      // Users can update activities if:
      // 1. They are the organizer (can update any field)
      // 2. They are joining/leaving (can only update participants array)
      allow update: if request.auth != null 
        && (
          // Organizer can update anything
          request.auth.uid == resource.data.organizerId
          ||
          // Non-organizers can only join/leave (update participants only)
          (
            request.resource.data.keys().hasAll(resource.data.keys()) &&
            resource.data.keys().hasAll(request.resource.data.keys()) &&
            request.resource.data.diff(resource.data).affectedKeys().hasOnly(['participants']) &&
            (
              // Joining - adding themselves to participants
              (
                request.resource.data.participants.hasAll(resource.data.participants) &&
                request.resource.data.participants.removeAll(resource.data.participants).hasOnly([request.auth.uid])
              )
              ||
              // Leaving - removing themselves from participants
              (
                resource.data.participants.hasAll(request.resource.data.participants) &&
                resource.data.participants.removeAll(request.resource.data.participants).hasOnly([request.auth.uid])
              )
            )
          )
        );
      
      // Only organizer can delete
      allow delete: if request.auth != null 
        && request.auth.uid == resource.data.organizerId;
    }

    // Chat messages
    match /chats/{activityId}/messages/{messageId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null 
        && request.auth.uid == request.resource.data.senderId;
      allow update, delete: if request.auth != null 
        && request.auth.uid == resource.data.senderId;
    }

    // Default deny all other collections
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
